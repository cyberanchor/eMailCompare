{{- /* Plain text from the CSV as safe HTML: escaped, URLs turned into links, line breaks kept. */ -}} {{- $h := htmlEscape (. | default "") -}} {{- $h = replaceRE `(https?://(?:[^\s<>&]|&)*[^\s<>&.,;:)\]])` `$1` $h -}} {{- $h = replace (strings.TrimSpace $h) "\n" "
" -}} {{- return ($h | safeHTML) -}}
