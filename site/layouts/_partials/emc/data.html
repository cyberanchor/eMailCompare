$host -}}{{- $net = "Facebook" -}} {{- else if eq $host "instagram.com" -}}{{- $net = "Instagram" -}} {{- else if findRE `linkedin\.com$` $host -}}{{- $net = "LinkedIn" -}} {{- else if findRE `youtube\.com$` $host -}}{{- $net = "YouTube" -}} {{- else if findRE `^threads\.` $host -}}{{- $net = "Threads" -}} {{- else if eq $host "tiktok.com" -}}{{- $net = "TikTok" -}} {{- else if in (slice
